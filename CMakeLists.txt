cmake_minimum_required(VERSION 3.25)
project(CPM VERSION 0.0.5.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(red4ext_sdk GIT_REPOSITORY https://github.com/WopsS/RED4ext.SDK.git GIT_TAG master GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(red4ext_sdk)
FetchContent_Declare(redlib GIT_REPOSITORY https://github.com/psiberx/cp2077-red-lib.git GIT_TAG master GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(redlib)
add_library(CPMClient SHARED src/Main.cpp src/Native.cpp src/Client.cpp src/Config.cpp src/Logger.cpp)
target_include_directories(CPMClient PRIVATE src)
target_compile_definitions(CPMClient PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CPMClient PRIVATE RED4ext.SDK RedLib ws2_32)
set_target_properties(CPMClient PROPERTIES OUTPUT_NAME "CPMClient")
add_executable(CPMServer server/Main.cpp)
target_include_directories(CPMServer PRIVATE src)
target_compile_definitions(CPMServer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CPMServer PRIVATE ws2_32)
